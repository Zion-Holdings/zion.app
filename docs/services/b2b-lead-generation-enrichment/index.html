Sample Composio Workflow: End-to-End Lead Processing

# When a new lead is discovered (Firecrawl trigger or manual import):
# 1. Research the company
tavily_search(query=f"{company} funding news recent")
perplexity_ai(prompt=f"Summarize {company}'s tech stack and recent initiatives")

# 2. Enrich & score
hubspot_create_contact(email, company, title, source)
airtable_update(record_id, {enrichment: research_data, score: ai_score})

# 3. If score > threshold:
gmail_create_draft(to=lead_email,
  subject=f"Re: {company}'s {recent_initiative}",
  body=personalized_outreach(research_data))
slack_send_message(channel="#sales-leads",
  text=f"🔥 Hot lead: {company} — score {score}/100 — {decision_maker}")

# 4. If no response in 3 days:
gmail_send_follow_up(lead_email, followup_template)
linkedin_send_connection_request(lead_profile)
